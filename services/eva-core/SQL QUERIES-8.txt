-- 1. Drop the old function to avoid conflicts
DROP FUNCTION IF EXISTS match_vectors;

-- 2. Recreate it with the Gemini dimension (768)
create or replace function match_vectors (
  query_embedding vector(768),   -- <--- CRITICAL CHANGE: 768 instead of 1536
  match_threshold float,
  match_count int,
  filter_tenant_id uuid
)
returns table (
  id uuid,
  text text,
  similarity float
)
language plpgsql stable
as $$
begin
  return query
  select
    vector_records.id,
    vector_records.text,
    1 - (vector_records.embedding <=> query_embedding) as similarity
  from vector_records
  where 1 - (vector_records.embedding <=> query_embedding) > match_threshold
  and tenant_id = filter_tenant_id
  order by similarity desc
  limit match_count;
end;
$$;
